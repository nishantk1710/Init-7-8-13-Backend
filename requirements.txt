# All Python dependencies for the backend, runtime and test alike.
#
# Keep this list minimal -- add a library only when the code actually imports
# it. Pin every version: an unpinned dependency makes a local pass and an Azure
# failure indistinguishable.
#
# Note: the test packages below are installed in every environment, deployment
# included. If a deployed image ever needs to stay lean, split the "Test" block
# back out into requirements-dev.txt with a leading `-r requirements.txt`.

# --- Runtime ---
fastapi==0.141.1
uvicorn[standard]==0.52.4
# The production server on App Service. uvicorn alone is a single process with
# no supervisor; gunicorn runs it under a master that replaces a worker which
# dies instead of taking the site down with it. See the startup command in
# README, "Deployment".
gunicorn==23.0.0

# Form and file-upload parsing, used by the CSV endpoint when SAP posts the
# extract as a multipart file part rather than as a raw body. Without it
# Starlette raises on such a request, and an upload would be lost to how it
# was wrapped rather than to anything in it.
python-multipart==0.0.20
pydantic-settings==2.15.0
# Imported directly by the API schemas and settings. Pinned rather than
# inherited from fastapi: a transitive major-version bump would land silently.
pydantic==2.13.5

# --- Data / persistence ---
# Azure SQL (sqldb-aicom) is the production backend. pyodbc additionally needs
# the system-level "ODBC Driver 18 for SQL Server", which is NOT a pip package
# -- it ships with the App Service Python image, and a local machine has to
# install it separately. That is the one dependency this file cannot pin for you.
SQLAlchemy==2.0.52
pyodbc==5.3.0
# Local dev and CI run against Postgres 18 instead (see compose.yaml and
# .github/workflows/ci.yml) -- psycopg is what DATABASE_URL's
# postgresql+psycopg:// scheme loads at migration/connect time.
psycopg[binary]==3.3.1
alembic==1.19.2
# Driver for the local Postgres stand-in (compose.yaml) -- DATABASE_URL there
# uses postgresql+psycopg://. Not needed against Azure SQL, but required for
# every local run; leaving it out makes every DB-touching endpoint 500.
psycopg[binary]==3.3.1

# --- Azure ---
# Data Lake storage (stvziaicomnonprod) behind the Storage port, and the managed
# identity that reaches it. azure-identity is what makes the deployed app need
# no storage secret at all.
azure-identity==1.19.0
azure-storage-file-datalake==12.17.0

# Reads the SAP extract workbooks. app/seed/reader.py opens every one of them
# with read_only=True -- see README, "Seeding".
openpyxl==3.1.5

# --- SAP / CPI ---
# HTTP to the CPI generic OData endpoint. Chosen over httpx because this is the
# library data-generator/cpi_discovery.py has been proven against for weeks --
# porting working auth and retry logic to a new client is unforced risk.
requests==2.34.2

# --- Forecasting (Initiative 07, Phase 4) ---
# SES and Auto-ARIMA. statsmodels rather than pmdarima: pmdarima is effectively
# unmaintained and does not build cleanly on Python 3.14, and the order search
# it provides is a loop over statsmodels fits that app/initiatives/i7/
# forecasting/arima.py does directly.
statsmodels==0.15.0

# The LightGBM quantile challenger for intermittent/lumpy demand. Trained as one
# global pooled model, never per material -- individual series here hold at most
# 13 observations.
lightgbm==4.7.0

# Pulled in by the two above; pinned so a fresh install matches this one.
numpy==2.5.3
scipy==1.18.1
pandas==3.0.5

# Reads config/dev/i7_service_level_matrix.yaml, the development-only mock
# Criticality -> Service Level matrix (see
# app/initiatives/i7/policy/dev_fixtures.py). Not used for anything else.
PyYAML==6.0.3

# --- Test ---
pytest==9.1.1
httpx2==2.12.0
